├── frontend/                        # React前端（Vite + TS）
│   ├── public/                      # 静态资源：favicon、静态html
│   ├── src/
│   │   ├── api/                     # axios实例封装、接口请求函数
│   │   ├── assets/                  # 图片、字体、全局样式文件
│   │   ├── components/              # 公共通用组件、业务封装组件
│   │   ├── hooks/                   # 自定义React Hooks
│   │   ├── layouts/                 # 页面布局组件（侧边栏、头部布局等）
│   │   ├── pages/                   # 业务页面，路由对应的页面
│   │   ├── router/                  # react-router路由配置
│   │   ├── store/                   # 状态管理
│   │   ├── types/                   # TS类型定义，对接后端接口
│   │   ├── utils/                   # 前端工具函数：格式化、校验等
│   │   ├── App.tsx
│   │   └── main.tsx
│   ├── .env.development             # 开发环境变量
│   ├── .env.production              # 生产环境变量
│   ├── package.json
│   ├── tsconfig.json
│   └── vite.config.ts
├── main.go                          # 唯一程序入口（Wails3 从根目录启动，调用 server/app）
├── go.mod                           # 根模块，通过 replace 引用 server 模块
├── Taskfile.yml                     # wails3 dev 的编译、Vite 与启动任务
├── build/config.yml                 # wails3 dev 的文件监视与执行配置
├── server/                          # Go Gin后端，go module: server
│   ├── app/
│   │   └── app.go                   # 启动装配：Gin 服务 + Wails 桌面窗口，headless 用构建标签隔离
│   ├── configs/
│   │   ├── config.example.yaml      # 配置模板，提交到git
│   │   └── config.yaml              # 本地真实配置，git忽略不上传
│   ├── docs/                        # swag自动生成的swagger接口文档
│   ├── go.mod
│   ├── go.sum
│   ├── Makefile                     # 脚本命令：run、build、migrate、swag生成
│   └── internal/
│       ├── bootstrap/               # 应用装配中心：初始化db、redis、日志、批量注册所有业务模块
│       ├── config/                  # 配置结构体定义、环境变量解析、环境变量覆盖配置
│       ├── middleware/              # Gin全局中间件：跨域、日志、jwt鉴权、租户解析、权限校验
│       ├── migrate/                 # Gorm自动迁移、数据表初始化、索引脚本
│       ├── model/                   # Gorm数据模型，一表一个文件，包含租户隔离字段
│       ├── modules/                 # 业务模块根目录，一个业务对应一个文件夹
│       │   ├── user/
│       │   │   ├── module.go        # 模块注册文件：实现Module接口，挂载路由、声明权限点
│       │   │   ├── handler.go       # 接口控制器，参数绑定、请求响应，Req/Resp结构体放这里
│       │   │   └── service.go       # 业务逻辑层，直接操作model，无repo层
│       ├── pkg/                     # 内部公共工具包，项目内通用
│       │   ├── errcode.go           # 统一错误码定义
│       │   ├── response.go          # 接口统一返回格式封装
│       │   ├── pagination.go        # 分页工具封装
│       │   ├── ctxkey.go            # context上下文key，存放租户、用户信息
│       │   └── jwtx.go              # jwt加解密工具
│       ├── router/                  # 路由底层抽象：Module接口定义、租户分组、权限包装器
│       └── tests/                   # 接口集成测试
├── .gitignore                       # git忽略文件配置
├── docker-compose.yml               # 容器编排：mysql、redis、前后端服务
└── README.md                        # 项目说明文档
